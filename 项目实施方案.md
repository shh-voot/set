## 🎯 问题分解与求解思路

### 问题一: 单架次最优装载

**输入:**
- 80个货箱 (重量、体积)
- 3种无人机 (载重、容积、航程)
- 调度中心到各服务区的3D距离

**目标:**
- 最大化单架次运输的物资价值/重量
- 满足重量、体积、能量约束

**算法:**
```
1. 数据预处理
   - 读取货箱数据 (重量w_i, 体积v_i)
   - 读取无人机参数 (载重W, 容积V, 电池E)
   - 计算每个服务区的往返能耗 (考虑3D地形)

2. 对每个服务区和每种机型:
   - 构建3D背包问题:
     max Σ w_i * x_i
     s.t. Σ w_i * x_i ≤ W
          Σ v_i * x_i ≤ V
          energy(route) ≤ E
          x_i ∈ {0, 1}
   
   - 求解方法:
     (a) 动态规划 (小规模)
     (b) 贪心+启发式 (大规模)
     (c) 按价值密度排序 (快速近似)

3. 输出最优装载方案
```

### 问题二: 多目标运输调度优化

**输入:**
- 80个货箱到15个服务区
- 3种机型，数量有限
- 电池数量和充电时间约束
- 4个优化目标

**目标函数:**
1. f₁: 最小化平均配送延迟
2. f₂: 最小化最大完成时间
3. f₃: 最小化总能耗
4. f₄: 最小化总架次数

**算法: 改进NSGA-II**

```
主框架:
---------
1. 编码设计 (6维染色体):
   [任务分组] [服务顺序] [机型分配] [无人机ID] [电池ID] [起飞时间]
   
   示例: 
   Groups:   [G1, G1, G2, G2, G3, ...]  # 哪些货箱一起运
   Order:    [S1, S3, S2, S5, ...]      # 服务区访问顺序
   Type:     [A, B, A, C, ...]          # 每架次用哪种机型
   UAV:      [1, 2, 1, 3, ...]          # 具体哪架无人机
   Battery:  [b1, b2, b3, b4, ...]      # 使用哪块电池
   Time:     [0, 5, 10, 15, ...]        # 起飞时刻 (分钟)

2. 初始化种群 (N=100-200):
   - Clarke-Wright算法生成初始解
   - 随机生成填充种群
   - 贪心启发式解

3. NSGA-II主循环 (G=500-2000代):
   for gen in 1..G:
     # 生成子代
     offspring = []
     for i in 1..N/2:
       p1, p2 = tournament_select(population, k=2)
       c1, c2 = crossover(p1, p2)
       c1, c2 = mutate(c1), mutate(c2)
       offspring += [c1, c2]
     
     # 评估目标函数
     evaluate(offspring)  # 并行评估加速
     
     # 合并父子代
     combined = population + offspring
     
     # 非支配排序
     fronts = fast_non_dominated_sort(combined)
     
     # 选择下一代
     population = []
     for front in fronts:
       if len(population) + len(front) <= N:
         crowding_distance(front)
         population += front
       else:
         crowding_distance(front)
         front.sort(key=lambda x: x.crowding_dist, reverse=True)
         population += front[:N - len(population)]
         break
   
   return pareto_front(population)

关键子函数:
-----------
def evaluate(individual):
    """评估一个解的4个目标"""
    # 解码染色体
    routes = decode(individual)
    
    # 目标1: 平均延迟
    f1 = sum(delivery_time[i] - deadline[i] for i in cargos) / len(cargos)
    
    # 目标2: 最大完成时间 (makespan)
    f2 = max(route.end_time for route in routes)
    
    # 目标3: 总能耗
    f3 = sum(route.energy for route in routes)
    
    # 目标4: 总架次数
    f4 = len(routes)
    
    return (f1, f2, f3, f4)

def decode(individual):
    """将染色体解码为实际执行方案"""
    routes = []
    battery_manager = BatteryManager(num_batteries, charge_power)
    
    for flight_idx in range(len(individual.groups)):
        # 获取任务信息
        cargos = individual.groups[flight_idx]
        service_area = individual.order[flight_idx]
        uav_type = individual.type[flight_idx]
        uav_id = individual.uav[flight_idx]
        battery_id = individual.battery[flight_idx]
        start_time = individual.time[flight_idx]
        
        # 检查约束
        if not check_feasibility(cargos, uav_type, service_area):
            continue  # 不可行解
        
        # 计算路线
        route = Route(
            uav_id=uav_id,
            uav_type=uav_type,
            cargos=cargos,
            destination=service_area,
            start_time=start_time,
            battery=battery_id
        )
        
        # 计算飞行时间和能耗
        route.compute_flight_profile(dem_data)
        
        # 电池管理
        if battery_manager.is_available(battery_id, start_time):
            battery_manager.assign(battery_id, route)
        else:
            # 等待充电或换电池
            wait_time = battery_manager.next_available_time(battery_id)
            route.start_time = wait_time
        
        routes.append(route)
    
    return routes

def crossover(parent1, parent2):
    """OX交叉算子"""
    # 服务区顺序交叉
    order1 = ox_crossover(parent1.order, parent2.order)
    order2 = ox_crossover(parent2.order, parent1.order)
    
    # 其他基因随机继承
    child1 = Individual(
        groups=random.choice([parent1.groups, parent2.groups]),
        order=order1,
        type=parent1.type[:len(order1)//2] + parent2.type[len(order1)//2:],
        uav=parent1.uav,
        battery=parent2.battery,
        time=parent1.time
    )
    
    return child1, child2

def mutate(individual):
    """多种变异算子"""
    if random.random() < 0.3:
        # 交换两个服务区顺序
        swap_mutation(individual.order)
    
    if random.random() < 0.2:
        # 改变机型分配
        type_mutation(individual.type)
    
    if random.random() < 0.1:
        # 调整起飞时间
        time_mutation(individual.time)
    
    return individual
```

```python
class BatteryManager:
    """电池充电调度管理器"""
    
    def __init__(self, num_batteries, charge_power_kw):
        self.batteries = [
            Battery(id=i, capacity_kwh=BATTERY_CAPACITY)
            for i in range(num_batteries)
        ]
        self.charge_power = charge_power_kw
        self.charge_events = []  # (battery_id, start, end)
    
    def is_available(self, battery_id, time):
        """检查电池在指定时间是否可用"""
        battery = self.batteries[battery_id]
        
        # 如果电池正在充电
        for event in self.charge_events:
            if event.battery_id == battery_id and event.end_time > time:
                return False
        
        # 如果电池电量足够
        return battery.soc >= 0.95
    
    def assign(self, battery_id, route):
        """分配电池给路线"""
        battery = self.batteries[battery_id]
        
        # 更新SOC
        energy_consumed = route.energy
        battery.soc -= energy_consumed / battery.capacity
        
        # 如果需要充电
        if battery.soc < 0.95:
            charge_start = route.end_time
            charge_duration = self.charging_time(
                battery.soc, 1.0, 
                battery.capacity, 
                self.charge_power
            )
            charge_end = charge_start + charge_duration
            
            self.charge_events.append(
                ChargeEvent(battery_id, charge_start, charge_end)
            )
    
    def charging_time(self, soc_start, soc_target, capacity, power):
        """两阶段充电模型 (来自文献06)"""
        FAST_THRESHOLD = 0.80
        TAPER_RATE = 0.4
        EFFICIENCY = 0.85
        
        energy_needed = capacity * (soc_target - soc_start)
        
        if soc_target <= FAST_THRESHOLD:
            # 全程快充
            return energy_needed / (power * EFFICIENCY)
        elif soc_start >= FAST_THRESHOLD:
            # 全程涓流
            avg_power = power * (1 + TAPER_RATE) / 2
            return energy_needed / (avg_power * EFFICIENCY)
        else:
            # 跨越两阶段
            fast_energy = capacity * (FAST_THRESHOLD - soc_start)
            taper_energy = capacity * (soc_target - FAST_THRESHOLD)
            
            t_fast = fast_energy / (power * EFFICIENCY)
            t_taper = taper_energy / (power * TAPER_RATE * EFFICIENCY)
            
            return t_fast + t_taper
```

```
def identify_blind_zones(routes, dem):
    """识别通信盲区"""
    blind_segments = []
    
    for route in routes:
        # 离散化路线为100个点
        path_points = discretize_route(route, num_points=100)
        
        for i, point in enumerate(path_points):
            # 检查该点到调度中心的通信
            if not has_line_of_sight(point, DEPOT, dem):
                # 记录盲区段
                blind_segments.append({
                    'route_id': route.id,
                    'segment_index': i,
                    'position': point,
                    'time': route.time_at_point(i)
                })
    
    return merge_continuous_segments(blind_segments)

def optimize_relay_placement(blind_segments):
    """贪心集合覆盖算法"""
    # 生成候选位置网格
    candidates = generate_candidate_grid(
        area_bounds=AREA_BOUNDS,
        grid_spacing=500,  # 500米间隔
        altitude_range=[300, 500, 700]  # 3个候选高度
    )
    
    uncovered = set(blind_segments)
    relays = []
    
    while uncovered:
        best_relay = None
        best_coverage = 0
        
        # 对每个候选位置
        for pos in candidates:
            # 计算覆盖多少盲区
            covered = set()
            for segment in uncovered:
                if can_relay_cover(pos, segment, dem):
                    covered.add(segment)
            
            # 更新最佳
            if len(covered) > best_coverage:
                best_coverage = len(covered)
                best_relay = pos
        
        if best_relay is None:
            break  # 无法完全覆盖
        
        relays.append(best_relay)
        uncovered -= covered
    
    return relays

def can_relay_cover(relay_pos, segment, dem):
    """检查中继能否覆盖某盲区段"""
    # 1. LOS检查
    los_to_depot = has_line_of_sight(relay_pos, DEPOT, dem)
    los_to_segment = has_line_of_sight(relay_pos, segment.position, dem)
    
    if not (los_to_depot and los_to_segment):
        return False
    
    # 2. 信号强度检查
    d1 = distance_3d(relay_pos, DEPOT)
    d2 = distance_3d(relay_pos, segment.position)
    
    power_depot = received_power_db(
        TX_POWER, TX_GAIN, RX_GAIN, d1, FREQ_MHZ
    )
    power_segment = received_power_db(
        TX_POWER, TX_GAIN, RX_GAIN, d2, FREQ_MHZ
    )
    
    return (power_depot > RX_THRESHOLD and 
            power_segment > RX_THRESHOLD)

def received_power_db(tx_power, tx_gain, rx_gain, distance, freq):
    """Friis公式 (dB形式)"""
    # 自由空间路径损耗
    fspl = 20 * log10(distance) + 20 * log10(freq) + 32.44
    
    # 接收功率
    rx_power = tx_power + tx_gain + rx_gain - fspl - SYSTEM_LOSS
    
    return rx_power
```

### 问题四: 服务区分组与资源配置

**目标:**
- 将15个服务区分成若干组
- 每组配置独立的调度中心和资源
- 分析资源缺口

**算法: 约束K-means聚类**

```
主流程:
-------
def cluster_service_areas(service_areas, num_clusters):
    """K-means聚类 + 约束优化"""
    
    # 1. 提取特征
    features = []
    for sa in service_areas:
        features.append([
            sa.longitude,
            sa.latitude,
            sa.altitude / 1000,  # 归一化
            sa.population / 1000,
            sa.cargo_weight / 100,
            sa.urgency_score
        ])
    
    # 2. K-means聚类
    from sklearn.cluster import KMeans
    kmeans = KMeans(n_clusters=num_clusters, random_state=42)
    labels = kmeans.fit_predict(features)
    
    # 3. 约束调整
    # 确保每组的总需求不超过单个调度中心的容量
    labels = adjust_for_capacity_constraints(labels, service_areas)
    
    # 4. 为每组分配资源
    groups = []
    for k in range(num_clusters):
        group_areas = [sa for i, sa in enumerate(service_areas) 
                       if labels[i] == k]
        
        # 计算资源需求
        resources = calculate_resource_requirements(group_areas)
        
        groups.append({
            'id': k,
            'service_areas': group_areas,
            'centroid': kmeans.cluster_centers_[k],
            'resources': resources
        })
    
    return groups

def calculate_resource_requirements(service_areas):
    """计算一组服务区需要的资源"""
    total_cargo_weight = sum(sa.cargo_weight for sa in service_areas)
    total_cargo_volume = sum(sa.cargo_volume for sa in service_areas)
    max_distance = max(sa.distance_to_centroid for sa in service_areas)
    
    # 估算需要的无人机数量 (基于问题二的结果)
    num_uav_A = estimate_uav_count('A', service_areas)
    num_uav_B = estimate_uav_count('B', service_areas)
    num_uav_C = estimate_uav_count('C', service_areas)
    
    # 估算电池需求
    num_batteries = max(num_uav_A, num_uav_B, num_uav_C) * 3  # 每架3块
    
    return {
        'uav_type_A': num_uav_A,
        'uav_type_B': num_uav_B,
        'uav_type_C': num_uav_C,
        'batteries': num_batteries,
        'charge_stations': max(2, num_batteries // 5)
    }

def analyze_resource_gap(actual_resources, required_resources):
    """分析资源缺口"""
    gaps = {}
    
    for resource_type in required_resources:
        required = required_resources[resource_type]
        actual = actual_resources.get(resource_type, 0)
        
        if actual < required:
            gaps[resource_type] = {
                'required': required,
                'actual': actual,
                'shortage': required - actual,
                'shortage_pct': (required - actual) / required * 100
            }
    
    return gaps
```
